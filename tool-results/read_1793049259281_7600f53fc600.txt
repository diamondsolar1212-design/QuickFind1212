    26→export function generateBusinessWebsite(business: BusinessData): string {
    27→  const tpl = business.template
    28→
    29→  // Parse products
    30→  let products: any[] = []
    31→  try { products = business.products ? JSON.parse(business.products) : [] } catch {}
    32→
    33→  // Parse gallery images
    34→  const gallery = business.imageUrls ? business.imageUrls.split(',').filter(Boolean) : []
    35→  const allImages = business.profileImage && !gallery.includes(business.profileImage)
    36→    ? [business.profileImage, ...gallery]
    37→    : gallery
    38→
    39→  // Parse hours
    40→  let hours: any = {}
    41→  try { hours = business.hours ? JSON.parse(business.hours) : {} } catch {}
    42→
    43→  switch (tpl) {
    44→    case 'calm-sage':
    45→      return generateCalmSage(business, allImages, products, hours)
    46→    case 'jungle-gold':
    47→      return generateJungleGold(business, allImages, products, hours)
    48→    case 'athletic-bold':
    49→      return generateAthleticBold(business, allImages, products, hours)
    50→    case 'tech-lime':
    51→      return generateTechLime(business, allImages, products, hours)
    52→    case 'journalism-red':
    53→      return generateJournalismRed(business, allImages, products, hours)
    54→    default:
    55→      return generateDefault(business, allImages, products, hours)
    56→  }
    57→}

    59→function escapeHtml(str: string): string {
    60→  return (str || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;')
    61→}
