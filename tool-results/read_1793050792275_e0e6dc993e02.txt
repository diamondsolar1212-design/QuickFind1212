    27→export function generateBusinessWebsite(business: BusinessData): string {
    28→  const tpl = business.template
    29→
    30→  // Parse products
    31→  let products: any[] = []
    32→  try { products = business.products ? JSON.parse(business.products) : [] } catch {}
    33→
    34→  // Parse gallery images
    35→  const gallery = business.imageUrls ? business.imageUrls.split(',').filter(Boolean) : []
    36→  const allImages = business.profileImage && !gallery.includes(business.profileImage)
    37→    ? [business.profileImage, ...gallery]
    38→    : gallery
    39→
    40→  // Parse hours
    41→  let hours: any = {}
    42→  try { hours = business.hours ? JSON.parse(business.hours) : {} } catch {}
    43→
    44→  switch (tpl) {
    45→    case 'calm-sage':
    46→      return generateCalmSage(business, allImages, products, hours)
    47→    case 'jungle-gold':
    48→      return generateJungleGold(business, allImages, products, hours)
    49→    case 'athletic-bold':
    50→      return generateAthleticBold(business, allImages, products, hours)
    51→    case 'tech-lime':
    52→      return generateTechLime(business, allImages, products, hours)
    53→    case 'journalism-red':
    54→      return generateJournalismRed(business, allImages, products, hours)
    55→    default:
    56→      return generateDefault(business, allImages, products, hours)
    57→  }
    58→}

    60→function escapeHtml(str: string): string {
    61→  return (str || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;')
    62→}

    65→function navBgStyle(b: BusinessData, fallback: string): string {
    66→  if (b.profileImage) {
    67→    return `background:linear-gradient(180deg,rgba(0,0,0,0.55) 0%,rgba(0,0,0,0.35) 100%),url('${b.profileImage}') center/cover fixed;`
    68→  }
    69→  return fallback

    72→// Star rating HTML (out of 5)
    73→function starsHtml(rating: number): string {
    74→  const full = Math.floor(rating)
    75→  const half = rating - full >= 0.5
    76→  let s = ''
    77→  for (let i = 0; i < full; i++) s += '★'
    78→  if (half) s += '☆'
    79→  for (let i = full + (half ? 1 : 0); i < 5; i++) s += '☆'
    80→  return s
    81→}
